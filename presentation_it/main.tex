\begin{frame}{Gas di Bose ideale}
  Il gas di Bose ideale è un sistema di $N$ bosoni \textbf{non interagenti}, descritto dalla meccanica statistica.
  \begin{equation*}
      \langle n_i \rangle = \frac{1}{e^{\beta(\epsilon_i - \mu)}-1}
  \end{equation*}
  \pause
  Il numero di occupazione dello stato fondamentale $N_0 = \langle n_0\rangle$ corrisponde alla condensazione. Transizione di fase a $T = T_c$.
  \begin{equation*}
      \frac{N_0}{N} = 1-\left(\frac{T}{T_c}\right)^\alpha \quad \text{for } T < T_c
  \end{equation*}
  In una scatola finita $\alpha = 3/2$, in una trappola armonica $\alpha = 3$.
\end{frame}

\begin{frame}{Equazione di Gross-Pitaevskii}
  Un sistema di \textbf{bosoni debolmente interagenti} può essere descritto con un approssimazione di campo medio, che restituisce la GPE:
  \begin{equation*}
      i\hbar \pdv{\psi(x,t)}{t} = \left[ 
          -\frac{\hbar^2}{2m}\nabla^2 + V(x,t) + g|\psi(x,t)|^2
      \right] \psi(x,t)
  \end{equation*}
  \pause
  Nel caso stazionario:
  \begin{equation*}
      \left[ 
          -\frac{\hbar^2}{2m}\nabla^2 + V(x) + g|\psi(x)|^2
      \right] \psi(x) = \mu \psi(x)
  \end{equation*}
  Quando l'interazione domina sul termine cinetico:
  \begin{equation*}
      n(x) = \frac{\mu - V(x)}{g} \quad \Rightarrow \quad R_{\rm TF} = \sqrt{\frac{2\mu}{m\omega^2}}
  \end{equation*}
\end{frame}

\begin{frame}{Miscele di spin a due componenti accoppiate}
  Le GPE contengono le costanti di interazione \colorbox{Dandelion}{intra} e \colorbox{Thistle}{inter}-componente interaction constants e l'\colorbox{lime}{accoppiamento} tra i due stati:
  \begin{equation*}
    \begin{aligned}
        &\left[ -\frac{\hbar^2}{2m}\nabla^2 + V(x) - \colorbox{lime}{$\dfrac{\delta_B}{2}$} + \colorbox{Dandelion}{$g_{\uparrow\uparrow}$}|\psi_\uparrow(x)|^2 + \colorbox{Thistle}{$g_{\uparrow\downarrow}$}|\psi_\downarrow(x)|^2
        \right] \psi_\uparrow(x) - \colorbox{lime}{$\dfrac{\hbar\Omega_R}{2}$}\psi_\downarrow(x) = \mu_\uparrow \psi_\uparrow(x) \\
        &\left[ -\frac{\hbar^2}{2m}\nabla^2 + V(x) + \colorbox{lime}{$\dfrac{\delta_B}{2}$} + \colorbox{Thistle}{$g_{\uparrow\downarrow}$}|\psi_\uparrow(x)|^2 + \colorbox{Dandelion}{$g_{\downarrow\downarrow}$}|\psi_\downarrow(x)|^2
        \right] \psi_\downarrow(x) - \colorbox{lime}{$\dfrac{\hbar\Omega_R}{2}$}\psi_\uparrow(x) = \mu_\downarrow \psi_\downarrow(x)
    \end{aligned}
  \end{equation*}
  % Depending on the values of $g_{aa}$, $g_{bb}$ and $g_{ab}$, the \textbf{ground state} of the system can behave in different manners
  \pause
  La magnetizzazione $Z = (n_\uparrow-n_\downarrow)/(n_\uparrow+n_\downarrow)$ produce un panorama energetico a doppia buca:
  \begin{equation*}
      E_{\rm MF}(Z) = -|\delta g|n Z^2 - 2\hbar\Omega_R \sqrt{1-Z^2} - 2\hbar\delta_{\rm eff} Z
      \label{eq:E_MF}
  \end{equation*}
\end{frame}

\begin{frame}{Conclusioni}
  Cosa abbiamo imparato?
  \begin{itemize}
    \item Il sistema mostra \textbf{proprietà diverse} tra interno ed esterno della bolla
    \item Larghezza dei domain wall \textbf{dipende dal coupling} $\Omega_R$
    \item Il fattore di crescita della bolla nel tempo è \textbf{independente} da $\Omega_R$
    \item Nella bolla, strutture periodiche \textbf{scompaiono} con la dimensione che aumenta. Al contrario, \textbf{appaiono} fuori dalla bolla.
    \item Lunghezza tipica di informazione all'esterno è legata alla \textbf{healing length} di Rabi
  \end{itemize}
  \pause
  Ricerca futura:
  \begin{itemize}
    \item Analisi del canale di \textbf{densità}
    \item Confronto con \textbf{simulazioni} numeriche di GPE 1D
    \item Comportamento a diverse \textbf{temperature}
  \end{itemize}
\end{frame}
